using namespace std;

int main() {
    int nilai[5] = {75, 88, 92, 60, 85};
    int total = 0;

    for (int i = 0; i <= 5; i++) {
        total += nilai[i];
    }

    double rata_rata = total / 5;
    cout << "Rata-rata = " << rata_rata << endl;
